\newcommand{\newbreakablebox}[3]{
	\ifcsundef{#1name}{%
		% define a default name if not defined before
		\tcbset{BreakableBox@title/.code n args={2}{\tcbset{title={use newcommand define #1name}}} }
	}{\relax}
	\DeclareTColorBox[auto counter,number within=section]{#1}{ g o t\label g }{
		common, % use common style
		#3, % use the style passed in
		IfValueTF={##1}
		{BreakableBox@title={#1}{##1}}
		{
			IfValueTF={##2}
			{BreakableBox@title={#1}{##2}}
			{BreakableBox@title={#1}{}}
		},
		IfValueT={##4}
		{
			IfBooleanTF={##3}
			{label={##4}}
			{VIVID@label={#2}{##4}}
		}
	}
	\DeclareTColorBox{#1*}{ g o }{
		common,#3,
		IfValueTF={##1}
		{BreakableBox@title={#1}{##1}}
		{
			IfValueTF={##2}
			{BreakableBox@title={#1}{##2}}
			{BreakableBox@title={#1}{}}
		},
	}
}

	\textbf{\Large 三角形面积与周长的二级结论笔记}
	\vspace{0.5em}
	\hrule
	\vspace{0.5em}
	
	\section{核心引理}
	在$\triangle ABC$中，已知角$A$的对边为$a$，则：
	\mynote{考试可直接用，等腰时取等号}
	
	\keyformula{
		S_{\Delta \max} = \dfrac{a^2}{4 \tan\dfrac{A}{2}}, \quad
		C_{\text{周max}} = a\left(1 + \frac{1}{\sin\dfrac{A}{2}}\right)
	}
	
	\subsection*{证明}
	1.  \textbf{面积最大值证明}\\
	由余弦定理：
	$
	a^2 = b^2 + c^2 - 2bc\cos A \ge 2bc(1 - \cos A)\Longrightarrow
	bc \le \dfrac{a^2}{2(1 - \cos A)}
	$
	
	三角形面积公式 $S_\Delta = \dfrac{1}{2}bc\sin A
	\le \dfrac{a^2\sin A}{4(1 - \cos A)}
	= \dfrac{a^2 \cdot 2\sin\dfrac{A}{2}\cos\dfrac{A}{2}}{4 \cdot 2\sin^2\dfrac{A}{2}}
	= \dfrac{a^2}{4\tan\dfrac{A}{2}}
	$
	
	当且仅当 $b = c$（即$\triangle ABC$为等腰三角形）时取等号。
	\mynote{关键：二倍角公式化简}
	
	\vspace{0.5em}
	2.  \textbf{周长最大值证明}\\
	由余弦定理变形：
	\[
	a^2 = (b + c)^2 - 2bc(1 + \cos A)
	\]
	结合均值不等式 $bc \le \left(\dfrac{b+c}{2}\right)^2$，得：
	\[
	a^2 \ge (b + c)^2 - \frac{(b + c)^2}{2}(1 + \cos A)
	= (b + c)^2 \cdot \sin^2\frac{A}{2}
	\]
	开方并整理周长 $C_{\text{周}} = a + b + c$：
	\[
	C_{\text{周}} \le a + \dfrac{a}{\sin\dfrac{A}{2}}
	= a\left(1 + \frac{1}{\sin\dfrac{A}{2}}\right)
	\]
	当且仅当 $b = c$（即$\triangle ABC$为等腰三角形）时取等号。
	
	\note{本引理也可通过正弦定理证明，取等条件均为$b=c$（等腰三角形），考试中可直接使用结论“秒杀”。}
	
	\hrule
	\vspace{0.5em}
	
	\section{例题应用}
\begin{liti}
	\textbf{题：}在$\triangle ABC$中，内角$A,B,C$的对边分别为$a,b,c$，且满足
	\[
	\dfrac{\sin B}{\sin A + \sin C} = \dfrac{a - c}{b + c}
	\]
	(1) 求角$A$；
	(2) 若$a = 2$，求$\triangle ABC$周长的取值范围。
	
	\tcblower \textbf{解答}
	1.  \textbf{求角$A$}\\
	由正弦定理，将角化边：
	\[
	\frac{b}{a + c} = \frac{a - c}{b + c}
	\]
	交叉相乘整理：
	\[
	b(b + c) = (a - c)(a + c) \implies b^2 + bc = a^2 - c^2
	\]
	即 $a^2 = b^2 + c^2 + bc$。
	对比余弦定理 $a^2 = b^2 + c^2 - 2bc\cos A$，得：
	\[
	\cos A = -\frac{1}{2}
	\]
	又 $0 < A < \pi$，故 $A = \dfrac{2\pi}{3}$。
	\note{余弦定理逆用，求特殊角}
	
	2.  \textbf{求周长取值范围}\\
	由引理，当$a=2$、$A=\dfrac{2\pi}{3}$时，周长最大值：
	\[
	C_{\text{周max}} = 2\left(1 + \dfrac{1}{\sin\dfrac{\pi}{3}}\right)
	= 2\left(1 + \dfrac{2}{\sqrt{3}}\right)
	= 2 + \dfrac{4\sqrt{3}}{3}
	\]
	再由三角形两边之和大于第三边：$b + c > a = 2$，故周长 $C_{\text{周}} = a + b + c > 4$。
	
	\[
		\text{周长取值范围：}\ 4 < C_{\text{周}} \le 2 + \frac{4\sqrt{3}}{3}
	\]
\end{liti}
	\note{周长下限由三角形三边关系直接得到，上限由引理结论快速求得，无需重复推导。}
	\mynote{下限易错，别忘三边关系！}
